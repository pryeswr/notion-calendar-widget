  <script>

    // State Variables

    let currentDate = new Date();

    let selectedDateStr = null;

    let events = JSON.parse(localStorage.getItem('notion_calendar_events') || '{}');


    // DOM Elements

    const monthLabel = document.getElementById('monthLabel');

    const yearLabel = document.getElementById('yearLabel');

    const calendarGrid = document.getElementById('calendarGrid');

    const prevBtn = document.getElementById('prevBtn');

    const nextBtn = document.getElementById('nextBtn');

    const todayBtn = document.getElementById('todayBtn');

    const themeToggleBtn = document.getElementById('themeToggleBtn');

    const themeIcon = document.getElementById('themeIcon');

    const activeNotesSummary = document.getElementById('activeNotesSummary');


    // Modal DOM Elements

    const eventModal = document.getElementById('eventModal');

    const modalContent = document.getElementById('modalContent');

    const closeModalBtn = document.getElementById('closeModalBtn');

    const modalDateTitle = document.getElementById('modalDateTitle');

    const modalDaySub = document.getElementById('modalDaySub');

    const eventsList = document.getElementById('eventsList');

    const addEventForm = document.addEventForm || document.getElementById('addEventForm');

    const eventInput = document.getElementById('eventInput');


    // Month Names

    const monthNames = [

      "January", "February", "March", "April", "May", "June", 

      "July", "August", "September", "October", "November", "December"

    ];


    // Initialize App

    function init() {

      setupTheme();

      renderCalendar();

      setupEventListeners();

      lucide.createIcons();

    }


    // Setup Theme Mode (System default or saved preference)

    function setupTheme() {

      const savedTheme = localStorage.getItem('notion_calendar_theme');

      if (savedTheme === 'dark' || (!savedTheme && window.matchMedia('(prefers-color-scheme: dark)').matches)) {

        document.documentElement.classList.add('dark');

        themeIcon.setAttribute('data-lucide', 'sun');

      } else {

        document.documentElement.classList.remove('dark');

        themeIcon.setAttribute('data-lucide', 'moon');

      }

      lucide.createIcons();

    }


    // Toggle Light/Dark Mode

    themeToggleBtn.addEventListener('click', () => {

      const isDark = document.documentElement.classList.toggle('dark');

      localStorage.setItem('notion_calendar_theme', isDark ? 'dark' : 'light');

      themeIcon.setAttribute('data-lucide', isDark ? 'sun' : 'moon');

      lucide.createIcons();

    });


    // Render Calendar Logic

    function renderCalendar() {

      const year = currentDate.getFullYear();

      const month = currentDate.getMonth();


      // Set Header Labels

      monthLabel.textContent = monthNames[month];

      yearLabel.textContent = year;


      // Get first day of month and total days in month

      const firstDayIndex = new Date(year, month, 1).getDay();

      const totalDays = new Date(year, month + 1, 0).getDate();

      const prevMonthTotalDays = new Date(year, month, 0).getDate();


      const today = new Date();

      const isCurrentMonth = today.getFullYear() === year && today.getMonth() === month;


      calendarGrid.innerHTML = '';


      // Previous Month Filler Days

      for (let i = firstDayIndex - 1; i >= 0; i--) {

        const dayNum = prevMonthTotalDays - i;

        const dayEl = createDayElement(dayNum, 'text-notion-lightMuted/40 dark:text-notion-darkMuted/30 cursor-default');

        calendarGrid.appendChild(dayEl);

      }


      // Current Month Days

      let totalNotesThisMonth = 0;

      for (let day = 1; day <= totalDays; day++) {

        const dateKey = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;

        const isToday = isCurrentMonth && day === today.getDate();

        const hasEvents = events[dateKey] && events[dateKey].length > 0;


        if (hasEvents) {

          totalNotesThisMonth += events[dateKey].length;

        }


        const dayEl = document.createElement('button');

        dayEl.className = `relative aspect-square flex flex-col items-center justify-center rounded-xl font-medium text-xs transition-all duration-150 focus:outline-none hover:bg-notion-lightBorder/60 dark:hover:bg-notion-darkBorder/60 ${

          isToday 

            ? 'bg-notion-accent text-white font-bold shadow-sm hover:bg-notion-accentDark' 

            : 'text-notion-lightText dark:text-notion-darkText'

        }`;


        dayEl.innerHTML = `

          <span>${day}</span>

          ${hasEvents ? `<span class="absolute bottom-1 w-1 h-1 rounded-full ${isToday ? 'bg-white' : 'bg-notion-accent'}"></span>` : ''}

        `;


        dayEl.addEventListener('click', () => openModal(dateKey, day, month, year));

        calendarGrid.appendChild(dayEl);

      }


      // Next Month Filler Days (to complete 6 rows = 42 cells total for consistent grid height)

      const totalGridCells = calendarGrid.children.length;

      const remainingCells = 42 - totalGridCells;

      for (let day = 1; day <= remainingCells; day++) {

        const dayEl = createDayElement(day, 'text-notion-lightMuted/40 dark:text-notion-darkMuted/30 cursor-default');

        calendarGrid.appendChild(dayEl);

      }


      // Update summary counter

      activeNotesSummary.textContent = `${totalNotesThisMonth} event${totalNotesThisMonth === 1 ? '' : 's'} this month`;

    }


    // Helper for generating non-interactive filler day elements

    function createDayElement(text, customClasses) {

      const el = document.createElement('div');

      el.className = `aspect-square flex items-center justify-center text-xs font-normal ${customClasses}`;

      el.textContent = text;

      return el;

    }


    // Open Event Modal

    function openModal(dateKey, day, month, year) {

      selectedDateStr = dateKey;

      const dateObj = new Date(year, month, day);

      

      const dayName = dateObj.toLocaleDateString('en-US', { weekday: 'long' });

      modalDateTitle.textContent = `${monthNames[month]} ${day}, ${year}`;

      modalDaySub.textContent = dayName;


      renderEventsList();


      eventModal.classList.remove('hidden');

      setTimeout(() => {

        eventModal.classList.remove('opacity-0');

        modalContent.classList.remove('scale-95');

      }, 10);


      eventInput.focus();

    }


    // Close Event Modal

    function closeModal() {

      eventModal.classList.add('opacity-0');

      modalContent.classList.add('scale-95');

      setTimeout(() => {

        eventModal.classList.add('hidden');

        selectedDateStr = null;

        eventInput.value = '';

      }, 200);

    }


    // Render Events inside Modal

    function renderEventsList() {

      eventsList.innerHTML = '';

      const dateEvents = events[selectedDateStr] || [];


      if (dateEvents.length === 0) {

        eventsList.innerHTML = `<p class="text-xs text-notion-lightMuted dark:text-notion-darkMuted italic text-center py-2">No notes for this date.</p>`;

        return;

      }


      dateEvents.forEach((evText, index) => {

        const evItem = document.createElement('div');

        evItem.className = 'flex items-center justify-between bg-notion-lightCard dark:bg-notion-darkBg border border-notion-lightBorder dark:border-notion-darkBorder rounded-lg px-2.5 py-1.5 text-xs group';

        

        evItem.innerHTML = `

          <span class="text-notion-lightText dark:text-notion-darkText break-all pr-2">${escapeHtml(evText)}</span>

          <button onclick="deleteEvent(${index})" class="text-notion-lightMuted hover:text-red-500 opacity-80 group-hover:opacity-100 transition-opacity p-0.5">

            <i data-lucide="trash-2" class="w-3.5 h-3.5"></i>

          </button>

        `;

        eventsList.appendChild(evItem);

      });


      lucide.createIcons();

    }


    // Add New Event

    addEventForm.addEventListener('submit', (e) => {

      e.preventDefault();

      const val = eventInput.value.trim();

      if (!val || !selectedDateStr) return;


      if (!events[selectedDateStr]) {

        events[selectedDateStr] = [];

      }


      events[selectedDateStr].push(val);

      localStorage.setItem('notion_calendar_events', JSON.stringify(events));


      eventInput.value = '';

      renderEventsList();

      renderCalendar();

    });


    // Delete Event

    window.deleteEvent = function(index) {

      if (!selectedDateStr || !events[selectedDateStr]) return;

      events[selectedDateStr].splice(index, 1);

      

      if (events[selectedDateStr].length === 0) {

        delete events[selectedDateStr];

      }


      localStorage.setItem('notion_calendar_events', JSON.stringify(events));

      renderEventsList();

      renderCalendar();

    };


    // Helper: Prevent XSS in HTML string

    function escapeHtml(text) {

      return text.replace(/[&<>"']/g, function(m) {

        return {

          '&': '&amp;',

          '<': '&lt;',

          '>': '&gt;',

          '"': '&quot;',

          "'": '&#039;'

        }[m];

      });

    }


    // Setup Event Listeners for Navigation

    function setupEventListeners() {

      prevBtn.addEventListener('click', () => {

        currentDate.setMonth(currentDate.getMonth() - 1);

        renderCalendar();

      });


      nextBtn.addEventListener('click', () => {

        currentDate.setMonth(currentDate.getMonth() + 1);

        renderCalendar();

      });


      todayBtn.addEventListener('click', () => {

        currentDate = new Date();

        renderCalendar();

      });


      closeModalBtn.addEventListener('click', closeModal);


      // Close modal on backdrop click

      eventModal.addEventListener('click', (e) => {

        if (e.target === eventModal) closeModal();

      });


      // Escape key to close modal

      document.addEventListener('keydown', (e) => {

        if (e.key === 'Escape' && !eventModal.classList.contains('hidden')) {

          closeModal();

        }

      });

    }


    // Initialize on load

    window.onload = init;

  </script>
